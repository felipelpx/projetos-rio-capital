⚙️ Setup — Mapas de Pagamento

Esta página cria as tabelas necessárias para a feature Mapas de Pagamento (aprovação 2 níveis).

Abre o Supabase SQL Editor e cola o SQL abaixo.

SQL — copia e cola tudo

-- 1. Tabela principal: cabeçalho do mapa
CREATE TABLE IF NOT EXISTS mapas_pagamento (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  numero text,                                    -- número incremental amigável (ex: MP-2026-001)
  descricao text,
  status text NOT NULL DEFAULT 'pendente_nivel_1',
    -- valores possíveis: pendente_nivel_1 | pendente_nivel_2 | aprovado | recusado
  total_valor numeric NOT NULL DEFAULT 0,
  created_by uuid REFERENCES profiles(id),
  created_at timestamptz NOT NULL DEFAULT now(),
  aprovado_n1_por uuid REFERENCES profiles(id),
  aprovado_n1_em timestamptz,
  aprovado_n2_por uuid REFERENCES profiles(id),
  aprovado_n2_em timestamptz,
  recusado_por uuid REFERENCES profiles(id),
  recusado_em timestamptz,
  motivo_recusa text
);

-- 2. Itens do mapa — referencia o tipo (fatura/pagamento/comissao) + id da origem
CREATE TABLE IF NOT EXISTS mapas_pagamento_itens (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  mapa_id uuid NOT NULL REFERENCES mapas_pagamento(id) ON DELETE CASCADE,
  tipo_origem text NOT NULL,                      -- 'fatura' | 'pagamento_extra' | 'comissao_sinal' | 'comissao_escritura' | 'recebivel_escritura'
  origem_id text NOT NULL,                        -- id da fatura, pagamento_extra ou venda
  descricao text NOT NULL,
  fornecedor text,
  empresa text,
  categoria text,
  valor numeric NOT NULL,
  vencimento date,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_mpi_mapa_id ON mapas_pagamento_itens(mapa_id);
CREATE INDEX IF NOT EXISTS idx_mp_status ON mapas_pagamento(status);

-- 3. Coluna de nível de aprovação nos perfis
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS approval_level int NOT NULL DEFAULT 0;
  -- 0 = sem aprovação, 1 = nível 1 (ex: Felipe), 2 = nível 2 (ex: Marcelo, Henrique, Marcia)

-- 4. Coluna "permite criar mapas" (qualquer admin pode por defeito; permite estender a outros utilizadores)
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS can_create_mapas boolean NOT NULL DEFAULT false;

-- 5. RLS (Row Level Security) — políticas
ALTER TABLE mapas_pagamento ENABLE ROW LEVEL SECURITY;
ALTER TABLE mapas_pagamento_itens ENABLE ROW LEVEL SECURITY;

-- Política mais simples: qualquer authenticated user vê e mexe (apertaremos depois se for preciso)
DROP POLICY IF EXISTS "auth full access mapas" ON mapas_pagamento;
CREATE POLICY "auth full access mapas" ON mapas_pagamento
  FOR ALL TO authenticated USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "auth full access mapas_itens" ON mapas_pagamento_itens;
CREATE POLICY "auth full access mapas_itens" ON mapas_pagamento_itens
  FOR ALL TO authenticated USING (true) WITH CHECK (true);

-- 6. Sequência para numeração amigável: MP-2026-001, MP-2026-002...
CREATE SEQUENCE IF NOT EXISTS mapas_pagamento_seq START 1;

Configurar os utilizadores

Depois de correres o SQL acima, configura os níveis de aprovação dos utilizadores:

-- Felipe = nível 1 (aprova primeiro)
UPDATE profiles SET approval_level = 1, can_create_mapas = true
  WHERE email = 'felipe@riocapital.pt';

-- Marcelo, Henrique, Marcia = nível 2 (aprovação final)
-- (substitui os emails pelos reais, se forem diferentes)
UPDATE profiles SET approval_level = 2, can_create_mapas = true
  WHERE email IN ('marcelo@riocapital.pt', 'henrique@riocapital.pt', 'marcia@riocapital.pt');

-- Thiago = pode criar mapas mas não aprova
UPDATE profiles SET approval_level = 0, can_create_mapas = true
  WHERE email = 'thiago@riocapital.pt';

-- Ver o resultado
SELECT email, role, approval_level, can_create_mapas FROM profiles ORDER BY email;
