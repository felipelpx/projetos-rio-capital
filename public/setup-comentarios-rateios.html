💬 Setup — comentários e divisão de movimentos

Duas funcionalidades novas precisam de tabelas que ainda não existem: os comentários nas linhas do extrato (movimento_comentarios) e a divisão de um movimento por várias categorias (movimento_rateios). Corre o SQL abaixo uma única vez — cria as duas.

1Abre o SQL Editor do Supabase

Vai a Supabase SQL Editor e cola o SQL abaixo.

2Corre o SQL

-- Comentários/perguntas em linhas do extrato
create table if not exists movimento_comentarios (
  id           uuid primary key default gen_random_uuid(),
  movimento_id text        not null,
  conta_id     text,
  texto        text        not null,
  autor_id     uuid,
  autor_nome   text,
  resolvido    boolean     not null default false,
  created_at   timestamptz not null default now()
);

create index if not exists idx_movcom_conta on movimento_comentarios(conta_id);
create index if not exists idx_movcom_mov   on movimento_comentarios(movimento_id);

alter table movimento_comentarios enable row level security;

-- Toda a equipa autenticada lê, escreve e resolve comentários.
drop policy if exists "ler comentarios"      on movimento_comentarios;
drop policy if exists "criar comentarios"    on movimento_comentarios;
drop policy if exists "editar comentarios"   on movimento_comentarios;
drop policy if exists "apagar comentarios"   on movimento_comentarios;

create policy "ler comentarios"    on movimento_comentarios for select using (true);
create policy "criar comentarios"  on movimento_comentarios for insert with check (true);
create policy "editar comentarios" on movimento_comentarios for update using (true) with check (true);
create policy "apagar comentarios" on movimento_comentarios for delete using (true);

-- Atualizações em tempo real (para o comentário aparecer sem recarregar)
do $$ begin
  if not exists (select 1 from pg_publication_tables
                 where pubname = 'supabase_realtime' and tablename = 'movimento_comentarios') then
    alter publication supabase_realtime add table movimento_comentarios;
  end if;
end $$;

-- ───────────────────────────────────────────────────────────────────────────
-- Divisão de um movimento por várias categorias (rateio)
create table if not exists movimento_rateios (
  id            uuid primary key default gen_random_uuid(),
  movimento_id  text        not null,
  conta_id      text,
  descricao     text,
  valor         numeric     not null,
  categoria     text,
  subcategoria  text,
  detalhes      text,
  created_at    timestamptz not null default now()
);

create index if not exists idx_rateio_conta on movimento_rateios(conta_id);
create index if not exists idx_rateio_mov   on movimento_rateios(movimento_id);

alter table movimento_rateios enable row level security;

drop policy if exists "ler rateios"    on movimento_rateios;
drop policy if exists "criar rateios"  on movimento_rateios;
drop policy if exists "editar rateios" on movimento_rateios;
drop policy if exists "apagar rateios" on movimento_rateios;

create policy "ler rateios"    on movimento_rateios for select using (true);
create policy "criar rateios"  on movimento_rateios for insert with check (true);
create policy "editar rateios" on movimento_rateios for update using (true) with check (true);
create policy "apagar rateios" on movimento_rateios for delete using (true);

do $$ begin
  if not exists (select 1 from pg_publication_tables
                 where pubname = 'supabase_realtime' and tablename = 'movimento_rateios') then
    alter publication supabase_realtime add table movimento_rateios;
  end if;
end $$;

notify pgrst, 'reload schema';

3Confirma

Volta aqui e carrega em Verificar. Depois abre o ERP → Extratos → uma conta: cada linha passa a ter um botão 💬 (notas) e um botão ⚖ (dividir por categorias).
